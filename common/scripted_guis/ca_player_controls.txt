# Direct player controls and native-policy bridge. The bridge is the only
# writer of native ACK lists; acknowledgements originate in actual GUI getters.

# An older ON save may load while paused without firing the startup on-action.
# This only requests the ordinary native handshake; it cannot acknowledge it.
ca_request_initial_knight_filters = {
	scope = character
	is_valid = {
		ca_player_maintenance_eligible_trigger = yes
		has_variable = ca_auto_knights_enabled
		NOT = { has_variable = ca_knight_filters_ready }
		OR = {
			NOT = { has_variable = ca_knight_reset_pending }
			NOT = { has_variable = ca_knight_reset_epoch }
		}
	}
	effect = {
		if = {
			limit = {
				ca_player_maintenance_eligible_trigger = yes
				has_variable = ca_auto_knights_enabled
				NOT = { has_variable = ca_knight_filters_ready }
				OR = {
					NOT = { has_variable = ca_knight_reset_pending }
					NOT = { has_variable = ca_knight_reset_epoch }
				}
			}
			ca_request_knight_filter_reset_effect = yes
			ca_schedule_maintenance_effect = yes
		}
	}
}

# The native worker supplies the captured transaction identity after its bounded
# timeout. A stale callback cannot disable a later request or another character.
ca_knight_filters_reset_failed = {
	scope = character
	is_valid = {
		ca_player_maintenance_eligible_trigger = yes
		has_variable = ca_auto_knights_enabled
		has_variable = ca_knight_reset_pending
		has_variable = ca_knight_reset_epoch
		exists = scope:ca_knight_ack_owner
		exists = scope:ca_knight_ack_epoch
		exists = scope:ca_knight_ack_local_player
		this = scope:ca_knight_ack_owner
		scope:ca_knight_ack_epoch = var:ca_knight_reset_epoch
		scope:ca_knight_ack_local_player = yes
	}
	effect = {
		if = {
			limit = {
				ca_player_maintenance_eligible_trigger = yes
				has_variable = ca_auto_knights_enabled
				has_variable = ca_knight_reset_pending
				has_variable = ca_knight_reset_epoch
				exists = scope:ca_knight_ack_owner
				exists = scope:ca_knight_ack_epoch
				exists = scope:ca_knight_ack_local_player
				this = scope:ca_knight_ack_owner
				scope:ca_knight_ack_epoch = var:ca_knight_reset_epoch
				scope:ca_knight_ack_local_player = yes
			}
			remove_variable = ca_auto_knights_enabled
			remove_variable = ca_auto_knights_initial_pending
			ca_next_knight_reset_epoch_effect = yes
			ca_automation_error_toast_effect = yes
		}
	}
}

ca_native_policy_begin = {
	scope = character
	is_valid = {
		is_ai = no
			is_landed = yes
			is_alive = yes
	}
	effect = {
		if = {
			limit = {
				is_ai = no
					is_landed = yes
					is_alive = yes
			}
			remove_variable = ca_native_policy_ready
			clear_variable_list = ca_native_off_confirmed
			clear_variable_list = ca_native_applied_confirmed
			set_variable = { name = ca_native_off_count value = 0 }
			set_variable = { name = ca_native_applied_count value = 0 }
			# Fixed counts for this owner/epoch: ignored roles contribute to neither.
			# Release ACKs shrink the pending list, but never this expected total.
			set_variable = { name = ca_native_type_count value = 0 }
			set_variable = { name = ca_native_apply_type_count value = 0 }
			every_court_position_type = {
				save_scope_as = ca_native_count_type
				root = {
					if = {
						limit = { ca_position_is_included_trigger = { POSITION = scope:ca_native_count_type } }
						change_variable = { name = ca_native_type_count add = 1 }
						if = {
							limit = { NOT = { has_variable = ca_auto_court_enabled } }
							change_variable = { name = ca_native_apply_type_count add = 1 }
						}
					}
					else_if = {
						limit = {
							is_target_in_variable_list = { name = ca_native_release_pending target = scope:ca_native_count_type }
							is_target_in_variable_list = { name = ca_native_saved_all target = scope:ca_native_count_type }
						}
						change_variable = { name = ca_native_apply_type_count add = 1 }
					}
				}
			}
			clear_saved_scope = ca_native_count_type
			if = {
				limit = { has_variable = ca_native_bridge_generation }
				change_variable = { name = ca_native_bridge_generation add = 1 }
			}
			else = { set_variable = { name = ca_native_bridge_generation value = 1 } }
			if = {
				limit = {
					NOT = { has_variable = ca_auto_court_enabled }
					OR = { has_variable = ca_bulk_automation_policy has_variable_list = ca_native_saved_all }
				}
				set_variable = { name = ca_native_restore_pending value = yes }
			}
		}
	}
}

ca_native_confirm_applied = {
	scope = character
	is_valid = {
		is_ai = no
			is_landed = yes
			is_alive = yes
		exists = scope:ca_native_type
		OR = {
			AND = {
				NOT = { has_variable = ca_auto_court_enabled }
				ca_position_is_included_trigger = { POSITION = scope:ca_native_type }
			}
			AND = {
				NOT = { ca_position_is_included_trigger = { POSITION = scope:ca_native_type } }
				is_target_in_variable_list = { name = ca_native_release_pending target = scope:ca_native_type }
				is_target_in_variable_list = { name = ca_native_saved_all target = scope:ca_native_type }
			}
		}
		NOT = { is_target_in_variable_list = { name = ca_native_applied_confirmed target = scope:ca_native_type } }
	}
	effect = {
		if = {
			limit = {
				is_ai = no
					is_landed = yes
					is_alive = yes
				exists = scope:ca_native_type
				OR = {
					AND = {
						NOT = { has_variable = ca_auto_court_enabled }
						ca_position_is_included_trigger = { POSITION = scope:ca_native_type }
					}
					AND = {
						NOT = { ca_position_is_included_trigger = { POSITION = scope:ca_native_type } }
						is_target_in_variable_list = { name = ca_native_release_pending target = scope:ca_native_type }
						is_target_in_variable_list = { name = ca_native_saved_all target = scope:ca_native_type }
					}
				}
				NOT = { is_target_in_variable_list = { name = ca_native_applied_confirmed target = scope:ca_native_type } }
			}
			add_to_variable_list = { name = ca_native_applied_confirmed target = scope:ca_native_type }
			change_variable = { name = ca_native_applied_count add = 1 }
			if = {
				limit = { is_target_in_variable_list = { name = ca_native_release_pending target = scope:ca_native_type } }
				# Only the actual native getter's ACK relinquishes ownership.
				remove_list_variable = { name = ca_native_release_pending target = scope:ca_native_type }
				ca_forget_native_position_policy_effect = { POSITION = scope:ca_native_type }
			}
		}
	}
}
